```bash
pnpm install --frozen-lockfile
pnpm typecheck
for test_file in tests/*.test.ts; do
  timeout 60s pnpm exec vitest run "$test_file" || exit $?
done
pnpm build
xvfb-run -a -s '-screen 0 3200x2000x24' pnpm test:ui
pnpm test:ime
pnpm benchmark
pnpm benchmark:research
pnpm benchmark:information
pnpm benchmark:strategies
pnpm benchmark:source-intent
pnpm benchmark:external

pnpm package:linux
```

复验成品与当前桌面：

```bash
POKE_EXECUTABLE="$PWD/release/0.3.0/linux-unpacked/poke-teambuilder" xvfb-run -a -s '-screen 0 3200x2000x24' pnpm test:ui
POKE_EXECUTABLE="$PWD/release/0.3.0/linux-unpacked/poke-teambuilder" pnpm test:ime
POKE_EXECUTABLE="$PWD/release/0.3.0/poke-teambuilder-0.3.0-x86_64.AppImage" xvfb-run -a -s '-screen 0 3200x2000x24' pnpm exec playwright test tests/desktop/artwork.spec.ts tests/desktop/history.spec.ts tests/desktop/configurations.spec.ts tests/desktop/environment.spec.ts tests/desktop/workspace.spec.ts --grep '雌性爱管侍|零样本赛季|应用升级|实际本地|核心构筑'
node tools/smoke-release.mjs
```

`pnpm benchmark` 需要参考目录 `../poke-type`，或设置 `POKE_REFERENCE_DIR`；应用本身不依赖旧工程。系统输入法检查需要 Xvfb、DBus、Fcitx5、Rime 和 `rime_deployer`。普通桌面检查使用临时数据目录，仅关闭本次启动的窗口。
